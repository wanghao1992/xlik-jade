# 天赋「究极武神霸斩」实现计划（炫酷特效版）

## 一、需求概述

新增一组 5 张逐级天赋卡（omni1 ~ omni5），复刻《最终幻想7》超究武神霸斩的单体处决连斩，并参考《DNF》剑魂「极·鬼剑术·暴风式」做**高密度剑暴风视觉**：

- 主动释放（占用主动天赋技能槽，与天降斩/剑阵/御剑一致，支持右键自动释放）
- 起手蓄势 → 目标周围形成**环绕剑阵/剑暴风** → 英雄带残影瞬移连斩 N 段 → **万剑归宗收束** → 巨剑终结斩 + 雷暴爆发 + 震屏
- 每段为「攻击力 × 倍率」物理伤害，走天赋机制伤害流程（带 `_extraDamage`，不触发分裂/闪电链等二次机制）
- omni2 ~ omni5 逐级提升：段数、每段倍率、终结倍率、终结范围、索敌范围、冷却、终结必暴
- 施法全程英雄无敌（`ability.invulnerable`）

## 三、机制与数值设计

### 3.1 战斗与视觉时间轴

| 阶段 | 时长 | 逻辑 | 视觉 |
| --- | --- | --- | --- |
| 起手蓄势 | 0.20s | 锁定最近敌人，快照参数，施加无敌 | 英雄脚下 `风爆.mdx`；`雷电光环.mdx` 绑定英雄 0.7s；目标周围生成 12 把紫蓝光「魔剑」环绕（剑暴风成型） |
| 连斩第 1~N 段 | 每段 0.24s | 目标死亡则重新索敌；`ability.moveStop` → `hero:position` 贴身 → `job.selectUnit` 补选 → 面向目标 → 结算 `攻击力 × 霸斩倍率` | 每段：离开点留 1 个英雄半透明残影（淡出 0.35s）；目标处 1 道 `天降斩-斩击` + 2 道随机旋转的 `T7_zhan_hit02`/`T4_hit5` 交叉刀光；剑暴风中心跟随当前目标 |
| 万剑归宗 | 0.18s | 停止风暴计时器 | 12 把剑向目标高速收束，半径/高度收缩、逐渐变淡，形成“收剑”前摇 |
| 终结斩 | 0.25s | 对主目标结算 `攻击力 × 霸斩终结倍率`（omni4 起必暴），对 `霸斩终结范围` 内其他敌人结算 50% | `狼骑-挥舞利刃.mdx`（2.6 倍、3 倍速）+ `风暴战锤.mdx` + `雷爆.mdx` + `球状闪电特效.mdx`；仅施法者本地 `camera.quake(3, 0.35)` |
| 收尾 | 0.25s | 清理状态、释放 `moveFlow`、销毁所有剑/残影/计时器 | 无残留特效 |

总时长 ≈ `0.20 + 段数×0.24 + 0.18 + 0.25`（omni5 十段约 3.0 秒）；无敌时长 = `段数×0.24 + 1.2` 秒。

--- 刀光爆发: 1 道弧形 + 2 道随机旋转交叉斩, 制造"乱舞"密度
---@param x number
---@param y number
---@param scale number
---@return void
local function omniSlashBurst(x, y, scale)
    local arc = effector.point(OMNI_SLASH_MODEL, x, y, nil, 0.28)
    if (type(arc) == "number") then
        effector.speed(arc, 6)
        effector.rotateZ(arc, math.rand(0, 359))
        effector.size(arc, scale * 1.2)
    end
    for _ = 1, 2 do
        local model = (math.rand(1, 2) == 1) and OMNI_SLASH_HIT_A or OMNI_SLASH_HIT_B
        local fx = effector.point(model, x, y, nil, 0.3)
        if (type(fx) == "number") then
            effector.speed(fx, 5)
            effector.rotateZ(fx, math.rand(0, 359))
            effector.size(fx, scale * math.rand(80, 120) / 100)
        end
    end
end

--- 英雄速度残影(复用凌波微步写法): 用英雄当前模型生成半透明虚影
---@param vfx table
---@param hero Unit
---@return void
local function omniTrail(vfx, hero)
    local model = hero:model()
    if (type(model) ~= "string" or #model == 0) then
        return
    end
    local e = effector.point(model, hero:x(), hero:y(), hero:z(), -1)
    if (nil == e) then
        return
    end
    effector.size(e, hero:modelScale() or 1)
    effector.rotateZ(e, hero:facing() or 270)
    effector.alpha(e, OMNI_TRAIL_ALPHA)
    effector.animation(e, 'walk', '')
    vfx.trails[#vfx.trails + 1] = { effect = e, life = OMNI_TRAIL_TIME }
end

--- 起手剑暴风: 12 把魔剑围绕目标高速环绕 + 上下起伏, 中心实时跟随当前目标
---@param vfx table
---@param tx number
---@param ty number
---@return void
local function omniStormStart(vfx, tx, ty)
    vfx.centerX, vfx.centerY = tx, ty
    local cz = japi.Z(tx, ty) + OMNI_STORM_HEIGHT
    for i = 1, OMNI_STORM_COUNT do
        local e = effector.agile(OMNI_STORM_SWORD, tx, ty, cz)
        if (effector.isAgile(e)) then

            speed = OMNI_STORM_SPEED * math.rand(80, 120) / 100,
        }
    end
    local elapsed = 0
    vfx.stormTimer = time.setInterval(OMNI_STORM_TICK, function()
        elapsed = elapsed + OMNI_STORM_TICK
        local cx = vfx.centerX or tx
        local cy = vfx.centerY or ty
        local baseZ = japi.Z(cx, cy)
        for _, s in ipairs(vfx.swords) do
            if (effector.isAgile(s.effect)) then

---@param tx number
---@param ty number

---@return void

        class.destroy(vfx.stormTimer)
        vfx.stormTimer = nil
    end
    local elapsed = 0
    local baseZ = japi.Z(tx, ty)
    vfx.convergeTimer = time.setInterval(OMNI_STORM_TICK, function()
        elapsed = elapsed + OMNI_STORM_TICK

            if (effector.isAgile(s.effect)) then
                local x, y = vector2.polar(tx, ty, s.radius * (1 - p), s.angle + p * 180)
                effector.position(s.effect, x, y, baseZ + s.height * (1 - p))

            vfx.convergeTimer = nil

--- 开始一套霸斩(参数在此快照)
---@param hero Unit
---@param firstTarget Unit
---@return boolean
local function startOmniSlash(hero, firstTarget)
    if (false == class.isObject(hero, UnitClass) or true == hero:isDead()) then
        return false
    end
    if (true == hero._talentOmniRunning) then
        return false
    end
    local owner = hero:owner()
    if (false == class.isObject(owner, PlayerClass)) then
        return false
    end
    local attr = hero._attr or {}
    local params = {
        segments = attr["霸斩段数"] or 5,
        ratio = attr["霸斩倍率"] or 0.35,
        search = attr["霸斩索敌"] or 900,
        finishRatio = attr["霸斩终结倍率"] or 1.50,
        finishRadius = attr["霸斩终结范围"] or 300,
        finishCrit = (attr["霸斩终结必暴"] or 0) > 0,
    }
    local base = hero:attack() or 1
    local target = firstTarget or omniFindTarget(hero, params.search, nil)
    if (false == class.isObject(target, UnitClass) or true == target:isDead()) then
        return false
    end

    hero._talentOmniRunning = true
    --- 全程无敌(段数/收束/终结耗时之上的兜底时长)
    ability.invulnerable({ whichUnit = hero, duration = params.segments * OMNI_TICK + 1.2 })

    --- VFX 状态
    local vfx = { swords = {}, trails = {}, centerX = target:x(), centerY = target:y() }
    hero._talentOmniVfx = vfx
    omniStormStart(vfx, target:x(), target:y())
    effector.attach(hero, OMNI_AURA_MODEL, "origin", 0.7)
    effector.point(OMNI_WINDUP_MODEL, hero:x(), hero:y(), nil, 0.5)
    vfx.trailTimer = time.setInterval(OMNI_TRAIL_TICK, function()
        for i = #vfx.trails, 1, -1 do
            local tr = vfx.trails[i]
            tr.life = tr.life - OMNI_TRAIL_TICK
            if (tr.life <= 0) then
                effector.destroy(tr.effect)
                table.remove(vfx.trails, i)
            else
                effector.alpha(tr.effect, math.floor(OMNI_TRAIL_ALPHA * tr.life / OMNI_TRAIL_TIME))
            end
        end
    end)

    local index, last = 0, nil

    local function finish()
        if (false == class.isDestroy(hero)) then
            hero._talentOmniRunning = nil
            hero._talentOmniVfx = nil
        end
        omniVfxStop(vfx)
        if (false == class.isDestroy(hero)) then
            moveFlow.release(hero, OMNI_LOCK_KEY)
        end
    end

    local function alive()
        return class.isObject(hero, UnitClass) and false == class.isDestroy(hero)
            and false == hero:isDead() and true == hero._talentOmniRunning
    end

    local function finisher()
        if (false == alive()) then
            finish()
            return
        end
        local t = (class.isObject(target, UnitClass) and false == target:isDead()) and target or last
        local tx, ty
        if (class.isObject(t, UnitClass) and false == t:isDead()) then
            tx, ty = t:x(), t:y()
        elseif (class.isObject(last, UnitClass)) then
            tx, ty = last:x(), last:y()
        else
            finish()
            return
        end
        local lx, ly = vector2.polar(tx, ty, OMNI_FINISH_OFFSET, math.rand(0, 359))
        omniTrail(vfx, hero)
        omniBlink(hero, owner, lx, ly, tx, ty)
        vfx.centerX, vfx.centerY = tx, ty
        --- 万剑归宗收束完成后: 爆发 + 伤害
        omniStormConverge(vfx, tx, ty, function()
            if (false == alive()) then
                finish()
                return
            end
            omniFinishBurst(owner, tx, ty)
            if (class.isObject(t, UnitClass) and false == t:isDead()) then
                job.attachEffect(t, OMNI_HIT_MODEL, "origin", 0.5)
                talentMechanicDamage(hero, t, base * params.finishRatio,
                    injury.damageType.physics, params.finishCrit)
            end
            for _, u in ipairs(enemiesAround(tx, ty, params.finishRadius, owner, t)) do
                job.attachEffect(u, OMNI_HIT_MODEL, "origin", 0.4)
                talentMechanicDamage(hero, u, base * params.finishRatio * OMNI_FINISH_SPLASH,
                    injury.damageType.physics, params.finishCrit)
            end
            time.setTimeout(0.25, finish)
        end)
    end

    local function step()
        if (false == alive()) then
            finish()
            return
        end
        if (index >= params.segments) then
            finisher()
            return
        end
        --- 目标死亡/丢失: 重新索敌(尽量换一个目标)
        if (false == class.isObject(target, UnitClass) or true == target:isDead()) then
            target = omniFindTarget(hero, params.search, last)
        end
        if (false == class.isObject(target, UnitClass) or true == target:isDead()) then
            finisher()
            return
        end
        index = index + 1
        local tx, ty = target:x(), target:y()
        local lx, ly = vector2.polar(tx, ty, OMNI_BLINK_OFFSET, math.rand(0, 359))
        omniTrail(vfx, hero)
        omniBlink(hero, owner, lx, ly, tx, ty)
        vfx.centerX, vfx.centerY = tx, ty
        omniSlashBurst(tx, ty, 1.2)
        omniHit(hero, target, base, params.ratio, false)
        last = target
        time.setTimeout(OMNI_TICK, step)
    end

    --- 起手蓄势后再进入第一段
    time.setTimeout(OMNI_WINDUP, step)
    return true
end

local function castOmniSlash(hero)
    if (true == hero._talentOmniRunning) then
        return false

        ttl = 2,
        maxDuration = 12,

            local vfx = (false == class.isDestroy(hero)) and hero._talentOmniVfx or nil

                if (false == class.isDestroy(hero)) then
                    hero._talentOmniVfx = nil
                end
                omniVfxStop(vfx)
            end
        end,
    })
    return true
end

## 五、假设与决策

1. **单体处决连斩**（用户已确认），不做范围风暴；终结斩附带小范围溅射以补足清群。
2. **主动技能槽**（用户已确认）：学会 omni1 后占 1 个主动天赋槽（上限 5），点击/热键释放，支持右键自动释放；`ready` 条件为索敌范围内有敌人。
3. **5 张新图标**（用户已确认），由 `war3-image-pipeline` 生成，命名 `天赋/omni1~5.blp`。
4. **炫酷特效（本轮新增）**：以「环绕剑暴风 + 英雄残影 + 万剑归宗 + 终结雷暴」为核心，全部复用已声明模型，不新增模型资源。
5. 震屏只作用于施法者本地屏幕（`async.call(owner, camera.quake)`），避免影响队友。
6. 伤害基数取 `hero:attack()`（与天降斩口径一致）；每段独立结算并带 `_extraDamage`，不触发分裂/闪电链/装备分裂。
7. 施法期间无敌，时长 `段数×0.24 + 1.2` 秒；被外部位移/传送强行打断时无敌会提前结束于 buff 自然到期，属于可接受的小余量。
8. 结束位置停在终结目标旁，不回到起手点（符合“处决”手感，且避免回拉与位移调度冲突）。
9. 全局魔法消耗沿用 `ACTIVE_MANA_COST = 5`；冷却逐级 45/42/40/37/35 秒。
10. 新卡自动进入三选一候选池，会稀释池子；不额外加权重/前置门槛，保持与其它机制卡一致。
11. 不新增模型、不改 `位移调度.lua`（priority 通过 `moveFlow.request` 显式传入 700）。

   - 学会后确认技能槽出现「究极武神霸斩」按钮（图标、名称、冷却转圈正常）。
3. 输入 `-tw` 直接测试（自动套用 omni5 满级参数），逐项核对视觉：
   - 起手：脚下风爆 + 英雄雷电光环，目标周围出现 12 把紫蓝飞剑高速环绕；
   - 连斩：英雄带半透明残影瞬移贴身，每段目标处有多道交叉刀光，剑暴风中心跟随目标；
   - 万剑归宗：飞剑向目标收束并变淡；
   - 终结：巨剑横扫 + 风暴战锤 + 雷爆 + 球状闪电，**只有施法者屏幕震动**；
   - 全程英雄无敌（怪物攻击无掉血），伤害数字按段跳出。
4. 结束后检查：
   - 英雄仍被选中，技能栏/热键正常（验证 `job.selectUnit`）；
   - 场上没有残留飞剑/残影，`-tq` 显示位移占用已释放。
5. 手动点击技能槽释放，确认有目标时成功、无目标时不扣蓝不进冷却。
6. 依次学会 omni2 ~ omni5，核对段数、每段伤害、终结伤害/范围、索敌、冷却是否随描述提升；omni4 起终结斩全部暴击。
7. 边界场景：
   - 斩击途中主目标死亡 → 自动切换到范围内下一个敌人，剑暴风中心随之移动；
   - 斩击途中所有敌人死亡 → 直接结束，无报错、无残留；
   - 连斩中用传送/天降斩强行打断，确认 `_talentOmniRunning` 与所有计时器/特效被清理；
   - 英雄在连斩中死亡 → 无报错、无残留特效。
8. 性能观察：多玩家同时释放时帧率稳定；若出现卡顿，优先下调 `OMNI_STORM_COUNT`（12→8）或每段交叉刀光数量（2→1）。
9. 回归：天降斩（`-tj`）、剑阵（`-tz`）、御剑（`-ty`）、剑气、天雷的冷却与位移互斥仍正常；同一段霸斩伤害不会触发二次机制。

## 八、风险与兜底

1. **特效密度/性能**：单次约 12 把风暴剑 + 最多 11 个残影 + 每段 3 道刀光 + 终结 4 个爆发，峰值并发约 20 个特效，与剑阵（3~5 秒内上百把小剑）同量级；若卡顿按第六节第 8 条降数量。
2. **瞬移选中闪烁**：每段 `position` 后调用 `job.selectUnit`；若实测观感不佳，可把 `OMNI_TICK` 提高到 0.3 秒或减少段数，无需改逻辑。
3. **位移冲突**：霸斩是 burst（priority 700），会挂起御剑/事件位移；`cancel` 清理状态与特效，`maxDuration` 看门狗兜底。
4. **震屏范围**：使用 `async.call` 本地震屏；若某些客户端在异步环境调用 `camera.quake` 异常，可包 `pcall` 或直接删掉震屏调用（不影响伤害逻辑）。
5. **数值过强/过弱**：所有数值集中在 `TALENT_EFFECTS` 的 7 个 `setExtra`，可直接调整；总倍率 810% 攻击力，若超标优先降 `霸斩倍率`。
6. **图标资源缺失**：若构建报缺图，检查 `vendor/assets/war3mapImage/天赋/omni*.blp` 是否生成，以及 `image_talent.lua` 是否声明。
